// Includes base do classificador Edge Impulse
#include "edge-impulse-sdk/classifier/ei_run_classifier.h"
#include "edge-impulse-sdk/classifier/ei_classifier_smooth.h"
#include "edge-impulse-sdk/dsp/numpy.hpp"

// Includes do modelo STRESS
#include "model-parameters-stress/model_metadata.h"
#include "model-parameters-stress/model_variables.h"

// Modelo TFLite compilado
#include "tflite_learn_790160_4_compiled.h"

// -----------------------------------------------------------------------------
// Função principal de inferência (renomeada de run_classifier)
// -----------------------------------------------------------------------------
namespace ei {

EI_IMPULSE_ERROR run_classifier_stress(
    ei::signal_t* signal,
    ei_impulse_result_t* result,
    bool debug
) {
    EI_IMPULSE_ERROR res = EI_IMPULSE_OK;
    TfLiteStatus tflite_status = kTfLiteOk;

    // Inicializar o modelo TFLite
    if (tflite_learn_790160_4_init(&ei::ei_malloc) != kTfLiteOk) {
        ei_printf("ERR: Falha ao inicializar o modelo STRESS\n");
        return EI_IMPULSE_TFLITE_ERROR;
    }

    // Obter ponteiro para o tensor de entrada
    TfLiteTensor input;
    if (tflite_learn_790160_4_input(0, &input) != kTfLiteOk) {
        ei_printf("ERR: Falha ao obter tensor de entrada\n");
        return EI_IMPULSE_TFLITE_ERROR;
    }

    // -------------------------------------------------------------------------
    // 1) Ler dados do sinal (features) e preencher o tensor de entrada
    // -------------------------------------------------------------------------
    size_t feature_count = EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE;
    for (size_t ix = 0; ix < feature_count; ix++) {
        float value;
        int r = signal->get_data(ix, 1, &value);
        if (r != 0) {
            ei_printf("ERR: Falha ao ler feature %zu\n", ix);
            return EI_IMPULSE_TFLITE_ERROR;
        }

#if EI_CLASSIFIER_TFLITE_INPUT_DATATYPE == EI_CLASSIFIER_DATATYPE_FLOAT32
        ((float*)input.data.data)[ix] = value;
#else
        ((int8_t*)input.data.data)[ix] =
            (int8_t)(value / input.params.scale + input.params.zero_point);
#endif
    }

    // -------------------------------------------------------------------------
    // 2) Executar inferência
    // -------------------------------------------------------------------------
    if (debug) ei_printf("Running STRESS model inference...\n");

    tflite_status = tflite_learn_790160_4_invoke();
    if (tflite_status != kTfLiteOk) {
        ei_printf("ERR: Falha ao executar o modelo STRESS\n");
        return EI_IMPULSE_TFLITE_ERROR;
    }

    // -------------------------------------------------------------------------
    // 3) Ler resultados do tensor de saída
    // -------------------------------------------------------------------------
    TfLiteTensor output;
    if (tflite_learn_790160_4_output(0, &output) != kTfLiteOk) {
        ei_printf("ERR: Falha ao obter tensor de saída\n");
        return EI_IMPULSE_TFLITE_ERROR;
    }

    size_t out_neurons = output.dims->data[1];
    float sum = 0.0f;

    // Converter outputs para float (caso quantizado)
    for (size_t i = 0; i < out_neurons; i++) {
        float value;
#if EI_CLASSIFIER_TFLITE_OUTPUT_DATATYPE == EI_CLASSIFIER_DATATYPE_FLOAT32
        value = ((float*)output.data.data)[i];
#else
        value = (output.data.int8[i] - output.params.zero_point) * output.params.scale;
#endif
        result->classification[i].value = value;
        result->classification[i].label = ei_classifier_inferencing_categories[i];
        sum += value;
    }

    // Normalizar (caso o modelo não tenha softmax interno)
    if (sum > 0.0f) {
        for (size_t i = 0; i < out_neurons; i++) {
            result->classification[i].value /= sum;
        }
    }

    // -------------------------------------------------------------------------
    // 4) Resultado final
    // -------------------------------------------------------------------------
    if (debug) {
        ei_printf("---- STRESS Inference Result ----\n");
        for (size_t i = 0; i < out_neurons; i++) {
            ei_printf("  %s: %.5f\n", result->classification[i].label,
                      result->classification[i].value);
        }
        ei_printf("---------------------------------\n");
    }

    // Reset e libertar memória
    tflite_learn_790160_4_reset(&ei::ei_free);

    return res;
}

} // namespace ei
